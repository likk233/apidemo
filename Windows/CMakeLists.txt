cmake_minimum_required(VERSION 3.20)
project(UsageBarWindows LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(UsageBarCore Core.cpp)
target_include_directories(UsageBarCore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(UsageBarChecks Tests.cpp)
target_link_libraries(UsageBarChecks PRIVATE UsageBarCore)
enable_testing()
add_test(NAME UsageBarChecks COMMAND UsageBarChecks)
if(WIN32)
  enable_language(RC)
  add_executable(UsageBar WIN32 Main.cpp Platform.cpp app.rc)
  target_include_directories(UsageBar PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(UsageBar PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(UsageBar PRIVATE UsageBarCore winhttp crypt32 shell32 ole32 uuid comctl32 uxtheme advapi32 gdi32 user32)
  add_executable(UsageBarPlatformChecks PlatformTests.cpp Platform.cpp)
  target_compile_definitions(UsageBarPlatformChecks PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(UsageBarPlatformChecks PRIVATE UsageBarCore winhttp crypt32 shell32 ole32 uuid advapi32)
  add_test(NAME UsageBarPlatformChecks COMMAND UsageBarPlatformChecks)
  if(MSVC)
    # Single EXE: statically link the compiler runtime; no VC++ redistributable required.
    set_property(TARGET UsageBar UsageBarCore UsageBarChecks UsageBarPlatformChecks PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(UsageBarCore PRIVATE /utf-8 /W4)
    target_compile_options(UsageBar PRIVATE /utf-8 /W4)
    target_compile_options(UsageBarChecks PRIVATE /utf-8 /W4)
    target_compile_options(UsageBarPlatformChecks PRIVATE /utf-8 /W4)
  elseif(MINGW)
    target_link_options(UsageBar PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(UsageBarChecks PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(UsageBarPlatformChecks PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
